export async function onRequest(context) { const url = new URL(context.request.url); const date = url.searchParams.get("date") || new Date().toISOString().split("T")[0]; const KEY = context.env.FOOTBALL_API_KEY; try { const res = await fetch(`https://live-football-api.com/api/v1/matches?api_key=${KEY}&date=${date}&lang=en`); const data = await res.json(); const raw = data.matches || data.data?.matches || data.data || data.result || []; const fixed = raw.map(m => { let kickoffTime = m.kickoff || "12:00"; let kickoffISO = `${date}T${kickoffTime}:00Z`; if (m.date && m.date.includes("T")) kickoffISO = m.date; let timestamp = Math.floor(new Date(kickoffISO).getTime() / 1000); let st = m.status || {}; let display = st.display || st.status || ""; if (typeof display === "object") display = display.display || ""; let s = String(display).toUpperCase(); let short = "NS"; if (s==="FT" || s.includes("FINISH")) short="FT"; else if (s.includes("HT")) short="HT"; else if (s.includes("1H") || s.includes("2H") || s==="LIVE" || st.is_live || m.is_live) short="LIVE"; let homeG = m.home?.score!=null? parseInt(m.home.score) : null; let awayG = m.away?.score!=null? parseInt(m.away.score) : null; let leagueName = (m.country && m.home)? m.name : (m.league?.name || m.name || "Other"); let country = m.country || ""; // elapsed will be null with this API, but frontend will show LIVE badge let minute = m.minute?? m.elapsed?? st.minute?? null; return { fixture: { id: m.id, date: kickoffISO, timestamp, status: { short, long: String(display), elapsed: minute } }, league: { name: leagueName, country }, teams: { home: { name: m.home?.name || "Home" }, away: { name: m.away?.name || "Away" } }, goals: { home: homeG, away: awayG } }; }); return new Response(JSON.stringify({response: fixed}), {headers: {"Content-Type":"application/json","Access-Control-Allow-Origin":"*","Cache-Control":"no-cache"}}); } catch(e){ return new Response(JSON.stringify({response:[],error:e.message}), {headers: {"Content-Type":"application/json","Access-Control-Allow-Origin":"*"}}); } }
