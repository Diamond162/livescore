export async function onRequest(context) { const url = new URL(context.request.url); const date = url.searchParams.get('date') || new Date().toISOString().split('T')[0]; const KEY = context.env.FOOTBALL_API_KEY; try { const r = await fetch(`https://live-football-api.com/api/v1/matches?api_key=${KEY}&date=${date}&lang=en`); const d = await r.json(); const raw = d.matches || d.data?.matches || d.data || d.result || []; const fixed = raw.map(m => { let kickoff = m.kickoff || m.date || m.time || `${date}T15:00:00Z`; if (typeof kickoff === 'number') kickoff = new Date(kickoff * 1000).toISOString(); if (typeof m.time === 'string' && m.time.length <= 5 && m.time.includes(':')) kickoff = `${date}T${m.time}:00+00:00`; let st = m.status; if (typeof st === 'object') st = st.name || st.type || ''; let s = String(st).toUpperCase(); let short = "NS"; if (s.includes("LIVE") || s.includes("INPLAY") || s==="1H" || s==="2H") short="LIVE"; else if (s.includes("FT") || s.includes("FINISH") || s.includes("ENDED")) short="FT"; return { fixture: { id: m.id, date: kickoff, status: { short, long: String(st) } }, league: { name: m.league?.name || m.competition?.name || "League", country: m.league?.country || "" }, teams: { home: { name: m.home?.name || m.home_name || "Home" }, away: { name: m.away?.name || m.away_name || "Away" } }, goals: { home: m.score?.home?? m.home_score?? null, away: m.score?.away?? m.away_score?? null } }; }); return new Response(JSON.stringify({ response: fixed }), { headers: { "Content-Type":"application/json", "Access-Control-Allow-Origin":"*" } }); } catch(e) { return new Response(JSON.stringify({ response:[], error: e.message }), { headers: { "Content-Type":"application/json" } }); } }
