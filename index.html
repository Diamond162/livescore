// ADD THIS MAP ON TOP OF YOUR FILE const COUNTRY_FLAGS = { 'INT': '🌍', 'INTL': '🌍', 'WORLD': '🌍', 'GER': '🇩🇪', 'GERMANY': '🇩🇪', 'ENG': '🏴󠁧󠁢󠁥󠁮󠁧󠁿', 'ENGLAND': '🏴󠁧󠁢󠁥󠁮󠁧󠁿', 'ESP': '🇪🇸', 'SPAIN': '🇪🇸', 'ITA': '🇮🇹', 'ITALY': '🇮🇹', 'FRA': '🇫🇷', 'FRANCE': '🇫🇷', 'NED': '🇳🇱', 'POR': '🇵🇹', 'TUR': '🇹🇷' }; function getCountryDisplay(league) { let code = (league.ccode || league.country || 'INT').toUpperCase(); let name = (league.country || 'INT').toUpperCase(); // Fix: INTL -> INT, GERMANY -> GER if (name.includes('CHAMPIONS') || name.includes('EUROPA') || name.includes('INT')) { code = 'INT'; name = 'INT'; } else if (name.includes('GERMAN')) { code = 'GER'; name = 'GER'; } const flag = COUNTRY_FLAGS[code] || COUNTRY_FLAGS[name] || '⚽'; return `${flag} ${code}: ${league.name.toUpperCase()}`; } // WHEN YOU RENDER LEAGUE HEADER, USE THIS: leagueHeader.innerHTML = getCountryDisplay(match.league);
